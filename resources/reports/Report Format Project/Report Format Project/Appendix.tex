\section*{Appendix A: Core Algorithmic Formulations}

\subsection*{Algorithm 1: Syntetos-Boylan Demand Classification \& Tournament Dispatch}
\begin{verbatim}
Input: Historical sales series Y = {y_1, y_2, ..., y_T}, Product ID p_id
Output: Demand Category, Best Model Name, 30-Day Forecast Vector, ROP, Q

1: Function ExecuteModelTournament(Y, p_id, LeadTime L=3, ServiceZ=1.65):
2:     positive_sales = [y for y in Y if y > 0]
3:     total_periods = Length(Y)
4:     non_zero_periods = Length(positive_sales)
5:     
6:     // Step 1: Syntetos-Boylan Categorization
7:     ADI = total_periods / non_zero_periods
8:     CV2 = (StdDev(positive_sales) / Mean(positive_sales)) ** 2
9:     
10:    If ADI < 1.32 and CV2 < 0.49 Then category = "Regular"
11:    Else If ADI < 1.32 and CV2 >= 0.49 Then category = "Erratic"
12:    Else If ADI >= 1.32 and CV2 < 0.49 Then category = "Intermittent"
13:    Else category = "Lumpy"
14:    
15:    // Step 2: Prune Candidate Model Pool
16:    If category == "Regular" Then
17:        candidates = ["Prophet", "Holt-Winters", "Gradient Boosting"]
18:    Else If category == "Erratic" Then
19:        candidates = ["Prophet", "Random Forest", "Gradient Boosting"]
20:    Else
21:        candidates = ["Simple Exponential Smoothing", "Random Forest"]
22:        
23:    // Step 3: Holdout Train/Test Split (80/20 Chronological)
24:    split_idx = Integer(total_periods * 0.8)
25:    train_data = Y[0 : split_idx]
26:    test_data = Y[split_idx : total_periods]
27:    
28:    // Step 4: Out-of-Sample Competition
29:    metrics = {}
30:    For model in candidates:
31:        y_pred = TrainAndPredict(model, train_data, Length(test_data))
32:        mae = Mean(AbsoluteValue(test_data - y_pred))
33:        metrics[model] = mae
34:        
35:    best_model = ModelWithMinimumMAE(metrics)
36:    forecast_30d = RetrainAndProject(best_model, Y, horizon=30)
37:    
38:    // Step 5: (s, S) Periodic Replenishment Calculations
39:    avg_daily_demand = Mean(forecast_30d)
40:    safety_stock = ServiceZ * StdDev(Y) * Sqrt(L)
41:    reorder_point = (avg_daily_demand * L) + safety_stock
42:    target_inventory = (avg_daily_demand * 7) + safety_stock
43:    current_stock = QueryInventory(p_id)
44:    
45:    order_qty = 0
46:    If current_stock < target_inventory and current_stock <= reorder_point:
47:        order_qty = Ceiling(target_inventory - current_stock)
48:        
49:    Return category, best_model, forecast_30d, reorder_point, order_qty
\end{verbatim}

\subsection*{Algorithm 2: Atomic POS Billing and Real-Time Stock Decrement}
\begin{verbatim}
Input: BillCreate Payload { items: List of [product_id, quantity] }, DB Session
Output: Persisted Bill Record, Synchronized Master Excel Ledger

1: Function ProcessCheckout(Payload, DB):
2:     product_ids = [item.product_id for item in Payload.items]
3:     
4:     // Begin Atomic Transaction
5:     products = DB.Query(Product).Filter(Product.id.In(product_ids)).WithForUpdate()
6:     product_map = {p.id: p for p in products}
7:     
8:     total_bill_amount = 0.0
9:     items_to_persist = []
10:    
11:    // Validation Phase
12:    For item in Payload.items:
13:        p = product_map.Get(item.product_id)
14:        If p is Null or item.quantity <= 0:
15:            Rollback(DB)
16:            Raise HTTPBadRequest("Invalid product or quantity")
17:        If p.quantity_in_stock < item.quantity:
18:            Rollback(DB)
19:            Raise HTTPConflict("Insufficient physical shelf stock")
20:            
21:        line_total = p.price * item.quantity
22:        total_bill_amount += line_total
23:        items_to_persist.Append((p, item.quantity, line_total))
24:        
25:    // Execution Phase
26:    bill = Bill(total_amount = total_bill_amount, created_at = SystemTimestamp())
27:    DB.Add(bill)
28:    DB.Flush() // Allocates unique bill.id
29:    
30:    For (p, qty, line_total) in items_to_persist:
31:        p.quantity_in_stock -= qty // Atomic in-memory decrement
32:        bill_item = BillItem(bill_id = bill.id, product_id = p.id,
33:                             product_name = p.name, quantity = qty,
34:                             unit_price = p.price, line_total = line_total)
35:        DB.Add(bill_item)
36:        
37:    DB.Commit() // Atomically flushes changes to SQLite
38:    
39:    // Asynchronous Audit Log
40:    AppendToMasterExcelWorkbook(bill, items_to_persist)
41:    
42:    Return bill
\end{verbatim}

\section*{Appendix B: Hardware and Runtime Environment Verification}
NexusERP operates completely within a self-contained local virtual environment. The system boots in under 2.5 seconds, consumes fewer than 450 MB of resident host RAM during multi-core model training, and requires 0 KB/s external network bandwidth for 100\% of POS and forecasting routines.
